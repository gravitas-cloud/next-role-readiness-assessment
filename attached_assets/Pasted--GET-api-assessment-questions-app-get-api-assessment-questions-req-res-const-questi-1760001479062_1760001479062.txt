// GET /api/assessment/questions
app.get('/api/assessment/questions', (req, res) => {
  const questions = assessmentContent.assessment_content.questions;
  const sections = assessmentContent.assessment_content.sections;
  
  res.json({
    opening: assessmentContent.assessment_content.opening_content,
    sections,
    questions
  });
});

// POST /api/assessment/submit
app.post('/api/assessment/submit', authenticateUser, async (req, res) => {
  const { responses } = req.body;
  
  // Validate all 12 questions answered
  const questionIds = ['q1', 'q2', 'q3', 'q4', 'q5', 'q6', 'q7', 'q8', 'q9', 'q10', 'q11', 'q12'];
  for (const qId of questionIds) {
    if (!responses[qId] || responses[qId] < 1 || responses[qId] > 5) {
      return res.status(400).json({ error: `Invalid response for ${qId}` });
    }
  }
  
  // Calculate results
  const results = calculateResults(responses);
  
  // Save to database
  const assessment = await prisma.assessment.create({
    data: {
      userId: req.user.id,
      responses: responses,
      strategicIdentity: results.strategicIdentity,
      visibilityPositioning: results.visibilityPositioning,
      politicalCapital: results.politicalCapital,
      totalScore: results.totalScore,
      zone: results.zone,
      priority1: results.priority1,
      priority2: results.priority2
    }
  });
  
  res.json({
    assessmentId: assessment.id,
    results
  });
});

// GET /api/assessment/:id/results
app.get('/api/assessment/:id/results', authenticateUser, async (req, res) => {
  const assessment = await prisma.assessment.findUnique({
    where: { id: req.params.id }
  });
  
  if (!assessment || assessment.userId !== req.user.id) {
    return res.status(404).json({ error: 'Assessment not found' });
  }
  
  // Get zone interpretation
  const zoneData = assessmentContent.assessment_content.zones[assessment.zone];
  const ctaContent = assessmentContent.assessment_content.cta_content;
  
  res.json({
    zone: zoneData,
    scores: {
      strategicIdentity: assessment.strategicIdentity,
      visibilityPositioning: assessment.visibilityPositioning,
      politicalCapital: assessment.politicalCapital,
      totalScore: assessment.totalScore
    },
    priorities: {
      priority1: assessment.priority1,
      priority2: assessment.priority2
    },
    cta: ctaContent
  });
});
